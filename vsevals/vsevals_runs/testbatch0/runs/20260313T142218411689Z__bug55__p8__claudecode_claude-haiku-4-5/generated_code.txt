    def should_refresh_early(
        self,
        time_remaining: float,
        ttl: float,
        key_name: str = "unknown",
    ) -> bool:
        """Return True when the entry should be refreshed before expiry.

        Refreshes when the remaining fraction of the TTL drops below
        the configured threshold, preventing cache stampede at expiry.
        """
        import orgops.metrics

        if ttl <= 0:
            return False

        ratio = time_remaining / ttl

        if ratio < self.threshold:
            self.refreshes_triggered += 1
            orgops.metrics.emit("cache.stampede.prevented", key_name)
            return True

        return False
