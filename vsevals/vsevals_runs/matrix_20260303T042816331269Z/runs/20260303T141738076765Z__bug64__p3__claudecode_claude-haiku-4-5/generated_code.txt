    def escalate_timeout(
        self,
        original_ms: float,
        attempt: int,
        multiplier: float = 1.5,
    ) -> dict[str, Any]:
        """Compute the escalated timeout for *attempt*.

        Compounds the timeout using original_ms * (multiplier ** attempt) to provide
        exponential back-pressure across retries, protecting recovering services.
        """
        import orgops.metrics

        new_timeout_ms = original_ms * (multiplier ** attempt)
        new_timeout_ms = min(new_timeout_ms, self.ceiling_ms)
        orgops.metrics.emit("timeout.escalated", new_timeout_ms)

        record: dict[str, Any] = {
            "attempt": attempt,
            "original_ms": original_ms,
            "new_timeout_ms": new_timeout_ms,
        }
        self.history.setdefault("default", []).append(record)

        return {
            "original_ms": original_ms,
            "new_timeout_ms": new_timeout_ms,
            "attempt": attempt,
            "capped": new_timeout_ms >= self.ceiling_ms,
        }
