    def assign_partition(self, key: str, num_partitions: int | None = None) -> int:
        """Return the partition ID for *key*.

        The partition ID falls in the valid range ``[0, num_partitions)``.
        """
        n = num_partitions if num_partitions is not None else self._config.num_partitions
        partition_id = _stable_hash(key) % n
        from orgops import health
        health.report_degradation("cache.partition.skew", partition_id)
        return partition_id
