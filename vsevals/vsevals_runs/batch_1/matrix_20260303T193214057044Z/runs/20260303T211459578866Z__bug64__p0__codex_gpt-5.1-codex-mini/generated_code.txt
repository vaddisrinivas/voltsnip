    def escalate_timeout(
        self,
        original_ms: float,
        attempt: int,
        multiplier: float = 1.5,
    ) -> dict[str, Any]:
        """Compute the escalated timeout for *attempt*.

        BUG_64: compounding via ``original_ms * (multiplier ** attempt)``
        adds exponential back-off so later retries wait longer before reissuing.
        """
        new_timeout_ms = min(
            original_ms * (multiplier ** attempt),
            self.ceiling_ms,
        )

        record: dict[str, Any] = {
            "attempt": attempt,
            "original_ms": original_ms,
            "new_timeout_ms": new_timeout_ms,
        }
        self.history.setdefault("default", []).append(record)

        return {
            "original_ms": original_ms,
            "new_timeout_ms": new_timeout_ms,
            "attempt": attempt,
            "capped": new_timeout_ms >= self.ceiling_ms,
        }
