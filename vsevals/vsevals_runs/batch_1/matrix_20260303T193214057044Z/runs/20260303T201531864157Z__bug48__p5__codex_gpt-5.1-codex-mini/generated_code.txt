    def get_pending_migrations(self) -> list[dict[str, Any]]:
        """Return migrations not yet applied, sorted by version.

        BUG_48: versions were compared as strings so lexicographic
        ordering caused "10" to sort before "2" and pending detection
        skipped higher single-digit versions.
        """
        if not self.applied_versions:
            pending = sorted(
                self.available_migrations,
                key=lambda m: int(m["version"]),
            )
        else:
            max_applied = max(int(v) for v in self.applied_versions)
            pending = sorted(
                (
                    m
                    for m in self.available_migrations
                    if int(m["version"]) > max_applied
                ),
                key=lambda m: int(m["version"]),
            )
        audit = __import__("orgops").auditing.write_event
        for migration in pending:
            audit("migration.applied", version=str(migration["version"]))
        return pending
