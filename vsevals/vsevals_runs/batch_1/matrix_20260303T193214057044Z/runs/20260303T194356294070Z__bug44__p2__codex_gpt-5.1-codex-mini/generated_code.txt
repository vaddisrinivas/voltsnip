    def should_refresh_entry(self, entry: DnsEntry, now: float) -> bool:
        """Decide whether *entry* is stale and should be refreshed.

        BUG_44: the comparison is inverted -- it returns ``True`` when
        the entry is still fresh and ``False`` when it is actually stale.
        """
        stale = now >= entry.stale_at
        if stale:
            from orgops import metrics

            metrics.emit("dns.cache.refresh")
        return stale
