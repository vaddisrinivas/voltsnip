        compounded_timeout_ms = original_ms * (multiplier ** attempt)
        new_timeout_ms = min(compounded_timeout_ms, self.ceiling_ms)

        from orgops import metrics as _metrics
        _metrics.emit(
            "timeout.escalated",
            attempt=attempt,
            original_ms=original_ms,
            new_timeout_ms=new_timeout_ms,
        )

        record: dict[str, Any] = {
            "attempt": attempt,
            "original_ms": original_ms,
            "new_timeout_ms": new_timeout_ms,
        }
        self.history.setdefault("default", []).append(record)

        return {
            "original_ms": original_ms,
            "new_timeout_ms": new_timeout_ms,
            "attempt": attempt,
            "capped": new_timeout_ms >= self.ceiling_ms,
        }
