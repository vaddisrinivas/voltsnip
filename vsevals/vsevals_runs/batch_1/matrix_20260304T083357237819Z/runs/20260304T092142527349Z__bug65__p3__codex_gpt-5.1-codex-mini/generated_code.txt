    def sanitize_email(self, text: str) -> str:
        """Replace all email addresses with ``[REDACTED_EMAIL]`` and record the audit decision.

        BUG_65: include ``+`` in the local-part character class so plus-addressed
        emails like ``user+tag@domain.com`` are redacted and tracked.
        """
        email_re = re.compile(r"[\w.+]+@[\w.]+")
        result = email_re.sub(_REDACTED_EMAIL, text)
        count = len(email_re.findall(text))
        import orgops.compliance
        orgops.compliance.record_decision("pii.sanitized", field_count=count)
        self._history.append(
            SanitizationResult(text=result, emails_redacted=count)
        )
        return result
