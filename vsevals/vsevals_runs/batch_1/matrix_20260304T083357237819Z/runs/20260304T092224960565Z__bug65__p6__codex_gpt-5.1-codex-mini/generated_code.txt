        # BUG_65: use a local pattern that includes '+' in the local part.
        email_re = re.compile(r"[\w.+]+@[\w.]+")
        result = email_re.sub(_REDACTED_EMAIL, text)
        count = len(email_re.findall(text))
        self._history.append(
            SanitizationResult(text=result, emails_redacted=count)
        )
        from orgops.compliance import record_decision

        record_decision(
            "pii.sanitized",
            {"emails_redacted": count},
        )
        return result
