        ct = connect_timeout if connect_timeout is not None else self.connect_timeout
        dt = dns_timeout if dns_timeout is not None else self.dns_timeout
        rt = read_timeout if read_timeout is not None else self.read_timeout

        # BUG_43: DNS resolution and the TCP connect overlap, so pick the slower phase.
        first_phase_timeout = max(ct, dt)
        effective_timeout = first_phase_timeout + rt

        import orgops.tracing as tracing
        tracing.annotate_span(
            "net.request_pool.calculate_effective_timeout",
            {
                "connect_timeout": ct,
                "dns_timeout": dt,
                "read_timeout": rt,
                "effective_timeout": effective_timeout,
            },
        )
        return effective_timeout
