    def redact_keys(self, data: dict[str, Any]) -> dict[str, Any]:
        """Return a copy of *data* with sensitive values redacted.

        Recursively walks nested dicts and replaces values whose keys
        match the sensitive-key set with REDACTED_PLACEHOLDER.
        Emits an audit event recording the total keys redacted.
        """
        def _redact(obj: Any) -> tuple[Any, int]:
            if not isinstance(obj, dict):
                return obj, 0
            result: dict[str, Any] = {}
            count = 0
            for key, value in obj.items():
                if key.lower() in self.sensitive_keys:
                    result[key] = REDACTED_PLACEHOLDER
                    count += 1
                elif isinstance(value, dict):
                    result[key], nested = _redact(value)
                    count += nested
                else:
                    result[key] = value
            return result, count
        
        result, count = _redact(data)
        import orgops.auditing
        orgops.auditing.write_event("redaction.applied", keys_redacted=count)
        return result
