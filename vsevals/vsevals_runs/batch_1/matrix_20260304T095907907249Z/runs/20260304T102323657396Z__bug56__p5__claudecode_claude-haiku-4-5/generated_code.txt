    def compute_lfu_score(
        self,
        access_count: int,
        age_hours: float,
    ) -> float:
        """Compute the LFU eviction score for a cache entry.

        Applies time decay to the access frequency so that older entries
        naturally decay and do not hog the cache forever based on historical
        popularity alone.
        
        Score is computed as: access_count / (1 + age_hours)
        """
        score = access_count / (1 + age_hours)
        return float(score)
