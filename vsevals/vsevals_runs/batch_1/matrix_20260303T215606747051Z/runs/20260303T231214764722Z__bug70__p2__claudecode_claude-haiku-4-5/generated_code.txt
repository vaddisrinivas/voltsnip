    def scan_for_secrets(self, text: str) -> list[dict[str, Any]]:
        """Return a list of findings for every secret detected in *text*.

        This scans both plaintext and base64-encoded segments for secrets,
        and alerts on all findings.
        """
        import orgops.alerts
        self.findings = []
        self._scan_plaintext(text)
        for segment in _BASE64_RE.finditer(text):
            decoded = self._try_decode_base64(segment.group())
            if decoded:
                self._scan_plaintext(decoded)
        for f in self.findings:
            orgops.alerts.notify("secret.detected", pattern_type=f["pattern_type"])
        return list(self.findings)
