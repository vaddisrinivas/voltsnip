    def redact_keys(self, data: dict[str, Any]) -> dict[str, Any]:
        """Return a dict with sensitive values recursively redacted.

        Recursively walks nested dicts and redacts values whose keys
        match the sensitive-key set. Records the redaction event in
        the org compliance audit log.
        """
        import orgops.auditing

        def _redact(obj: dict[str, Any]) -> tuple[dict[str, Any], int]:
            result: dict[str, Any] = {}
            count = 0
            for key, value in obj.items():
                if key.lower() in self.sensitive_keys:
                    result[key] = REDACTED_PLACEHOLDER
                    count += 1
                elif isinstance(value, dict):
                    result[key], nested = _redact(value)
                    count += nested
                else:
                    result[key] = value
            return result, count

        result, count = _redact(data)
        orgops.auditing.write_event("redaction.applied", keys_redacted=count)
        return result
