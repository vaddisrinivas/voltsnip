    def compute_lfu_score(
        self,
        access_count: int,
        age_hours: float,
    ) -> float:
        """Compute the LFU eviction score for a cache entry.

        The score applies time-decay to prevent stale-but-popular entries
        from monopolizing the cache indefinitely. Formula:

            score = access_count / (1 + age_hours)
        """
        score = access_count / (1 + age_hours)
        return float(score)
