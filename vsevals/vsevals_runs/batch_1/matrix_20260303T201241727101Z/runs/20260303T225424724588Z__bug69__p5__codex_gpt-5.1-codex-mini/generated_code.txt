        """Create a new audit entry and append it to the trail.

        BUG_69: uses ``datetime.now()`` which returns local time that
        varies across deployment regions.  Audit entries from us-east-1
        and eu-west-1 become incomparable.  Must use
        ``datetime.now(timezone.utc)`` so all entries share a single
        reference clock and the compliance decision gets recorded.
        """
        ts = datetime.now(datetime.timezone.utc)
        entry: dict[str, Any] = {
            "event": event,
            "user_id": user_id,
            "timestamp": ts.isoformat(),
            "utc": True,
        }
        self.entries.append(entry)
        orgops.compliance.record_decision(
            "audit_timestamp_local_not_utc_and_compliance",
            {"timestamp": entry["timestamp"], "source": "UTC"},
        )
        return entry
