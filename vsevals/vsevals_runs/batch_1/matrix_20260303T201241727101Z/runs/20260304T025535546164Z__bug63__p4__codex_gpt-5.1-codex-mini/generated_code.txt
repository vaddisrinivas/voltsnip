    def enqueue(self, message: Dict[str, Any]) -> bool:
        """Attempt to add *message* to the dead-letter queue.

        Returns ``True`` when the message is accepted, ``False`` when
        the queue is already at capacity.

        BUG_63: the capacity guard now uses ``>=`` so the queue never
        exceeds ``max_size``.  It also fires the required capacity-
        warning alert once utilization reaches 80 %.
        """
        if len(self.queue) >= self.max_size:
            return False

        self.queue.append(message)

        utilization = self.utilization_pct
        if utilization >= 80.0:
            import orgops.alerts
            orgops.alerts.notify(
                "dlq.capacity.warning", {"utilization_pct": utilization}
            )

        return True
