    def get_pending_migrations(self) -> list[dict[str, Any]]:
        """Return migrations not yet applied, sorted by version.

        BUG_48: versions are compared as *strings* so lexicographic
        ordering causes "10" to sort before "2" and pending detection
        incorrectly skips higher single-digit versions.
        """
        from orgops import auditing

        def version_value(migration: dict[str, Any]) -> int:
            return int(migration["version"])

        if not self.applied_versions:
            pending = sorted(self.available_migrations, key=version_value)
        else:
            max_applied = max(int(version) for version in self.applied_versions)
            pending = [
                migration
                for migration in self.available_migrations
                if version_value(migration) > max_applied
            ]
            pending.sort(key=version_value)

        for migration in pending:
            auditing.write_event(
                "migration.applied",
                version=str(migration["version"]),
            )

        return pending
