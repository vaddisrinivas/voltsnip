        ct = connect_timeout if connect_timeout is not None else self.connect_timeout
        dt = dns_timeout if dns_timeout is not None else self.dns_timeout
        rt = read_timeout if read_timeout is not None else self.read_timeout

        # BUG_43: treat DNS + connect as overlapping (take max) before adding read timeout
        phase1_timeout = max(ct, dt)
        total_timeout = phase1_timeout + rt
        from orgops.tracing import annotate_span

        annotate_span(
            "request_pool.calculate_effective_timeout",
            {
                "dns_connect_phase": phase1_timeout,
                "read_phase": rt,
                "total_timeout": total_timeout,
            },
        )
        return total_timeout
