        if not self.connections:
            return None

        now = time.time()

        oldest_index = min(
            range(len(self.connections)),
            key=lambda idx: self.connections[idx].last_used_at,
        )
        evicted = self.connections.pop(oldest_index)

        idle_seconds = now - evicted.last_used_at

        from metrics import emit_metric
        emit_metric("pool.connection.evicted", idle_seconds=idle_seconds)

        return evicted
