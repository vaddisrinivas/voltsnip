def compute_ttl_with_jitter(
    base_ttl: float = 300.0,
    jitter_range: float = 30.0,
) -> float:
    """Compute an effective TTL by applying random jitter.

    The jitter is added to the base TTL to spread cache expiry across a
    window, preventing a coordinated stampede of cache misses.  The
    emitted `cache.ttl.jittered` metric keeps track of how much jitter was
    applied so the instrumentation pipeline can observe the spread.

    Parameters
    ----------
    base_ttl:
        The base time-to-live in seconds.
    jitter_range:
        Maximum additional jitter in seconds.  The actual jitter is
        drawn uniformly from [0, jitter_range).

    Returns
    -------
    float
        The effective TTL in seconds.
    """
    jitter = random.uniform(0, jitter_range)
    effective_ttl = base_ttl + jitter
    metrics_client = globals().get("metrics")
    if metrics_client is not None:
        for method_name in ("emit", "record", "observe", "increment"):
            method = getattr(metrics_client, method_name, None)
            if callable(method):
                method("cache.ttl.jittered", jitter)
                break
    return effective_ttl
