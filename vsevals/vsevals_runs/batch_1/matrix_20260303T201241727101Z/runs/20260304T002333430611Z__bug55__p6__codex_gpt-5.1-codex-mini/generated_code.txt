    def should_refresh_early(
        self,
        time_remaining: float,
        ttl: float,
        key_name: str = "unknown",
    ) -> bool:
        """Return True when the entry should be refreshed before expiry.

        BUG_55: the ratio comparison is inverted, causing early refreshes
        when the entry is FAR from expiry (most of the TTL left) rather
        than CLOSE to expiry (little TTL left).
        """
        if ttl <= 0:
            return False

        ratio = time_remaining / ttl

        # BUG_55: refresh when ratio is LOW (close to expiry) instead of HIGH
        if ratio < self.threshold:
            self.refreshes_triggered += 1
            try:
                from cache import metrics
            except ImportError:
                pass
            else:
                for emitter in ("emit", "record", "increment"):
                    metric_call = getattr(metrics, emitter, None)
                    if callable(metric_call):
                        metric_call("cache.stampede.prevented")
                        break
            return True

        return False
