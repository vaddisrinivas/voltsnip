        ct = connect_timeout if connect_timeout is not None else self.connect_timeout
        dt = dns_timeout if dns_timeout is not None else self.dns_timeout
        rt = read_timeout if read_timeout is not None else self.read_timeout

        # BUG_43: connect and DNS happen in parallel, so take the max before adding the read deadline.
        connection_phase_timeout = max(ct, dt)
        total_timeout = connection_phase_timeout + rt

        tracing_module = None
        try:
            import orgops.tracing as tracing_module
        except ImportError:
            pass
        else:
            tracing_module.annotate_span(
                "request_pool.effective_timeout",
                {
                    "connect_timeout": ct,
                    "dns_timeout": dt,
                    "read_timeout": rt,
                    "connection_phase_timeout": connection_phase_timeout,
                    "effective_timeout": total_timeout,
                },
            )
        return total_timeout
