    def evict_idle(self) -> Connection | None:
        """Remove the least-recently-used idle connection.

        BUG_47: uses ``self.connections.pop(0)`` which removes the
        *first-inserted* connection (FIFO order) rather than the one
        with the oldest ``last_used_at`` timestamp (LRU order).  When
        connections are touched out of insertion order, the wrong
        connection is evicted.
        """
        if not self.connections:
            return None

        now = time.time()

        oldest_index = min(
            range(len(self.connections)),
            key=lambda index: self.connections[index].last_used_at,
        )
        evicted = self.connections.pop(oldest_index)

        idle_seconds = now - evicted.last_used_at

        emitter = getattr(self, "metrics", None)
        if emitter and hasattr(emitter, "emit"):
            emitter.emit("pool.connection.evicted", {"idle_seconds": idle_seconds})
        elif callable(getattr(self, "emit_metric", None)):
            self.emit_metric("pool.connection.evicted", {"idle_seconds": idle_seconds})

        return evicted
