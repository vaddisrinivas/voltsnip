    def redact_keys(self, data: dict[str, Any]) -> dict[str, Any]:
        """Return a copy of *data* with sensitive values redacted and audit the scrub count."""
        count = 0

        def _redact_dict(payload: dict[str, Any]) -> dict[str, Any]:
            nonlocal count
            redacted: dict[str, Any] = {}
            for key, value in payload.items():
                if self.is_sensitive(key):
                    redacted[key] = REDACTED_PLACEHOLDER
                    count += 1
                elif isinstance(value, dict):
                    redacted[key] = _redact_dict(value)
                else:
                    redacted[key] = value
            return redacted

        result = _redact_dict(data)
        import orgops

        orgops.auditing.write_event("redaction.applied", keys_redacted=count)
        return result
