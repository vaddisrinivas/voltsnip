    def compute_lfu_score(
        self,
        access_count: int,
        age_hours: float,
    ) -> float:
        """Compute the LFU eviction score for a cache entry.

        BUG_56: returns the raw *access_count* without any time-decay
        factor.  The correct formula is:

            score = access_count / (1 + age_hours)

        so that older entries naturally decay and do not hog the cache
        forever based on historical popularity alone.
        """
        score = access_count / (1 + age_hours)
        try:
            from cache import metrics as _metrics
        except ImportError:
            _metrics = None
        if _metrics is not None:
            if hasattr(_metrics, "gauge"):
                _metrics.gauge("cache.eviction.lfu", value=score)
            elif hasattr(_metrics, "emit"):
                _metrics.emit("cache.eviction.lfu", value=score)
            elif hasattr(_metrics, "record"):
                _metrics.record("cache.eviction.lfu", value=score)
        return float(score)
