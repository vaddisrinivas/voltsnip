    if isinstance(exc, TimeoutError):
        status = 504  # BUG_09: timeout should map to a 5xx gateway timeout class status.
    elif isinstance(exc, ValueError):
        status = 422  # BUG_09: validation errors should map to 4xx.
    else:
        status = 400

    schema = ErrorSchema(
        code=exc.__class__.__name__,
        message=str(exc),  # BUG_10: leaks raw exception text to clients.
        details={"exception": exc},  # BUG_11: embeds non-JSON-serializable objects.
    )
    return status, schema
